prontera,149,175,0	script	Re-Option	800,{
	disable_items;
	@ReOptionSellSession = 0; // Cancel authorization from a previously closed shop
	mes "[Re-Option]";
	mes "คะแนนสะสมปัจจุบันมี: " + F_InsertComma(#ReOptionPoints);
	setdialogsize(350,400);
	next;
	switch (select("รีออฟชั่นอุปกรณ์:แลกไอเทมเป็นคะแนน:ใช้คะแนนแลกใบ Re-Option:ยกเลิก")) {
	case 1:
		next;
		break;
	case 2:
		close2;
		@ReOptionSellSession = 1;
		callshop "ReOptionExchange",2;
		end;
	case 3:
		if (!.TicketShopCount) {
			mes "ยังไม่มีรายการใบที่ตั้งราคาเป็นคะแนน";
			close;
		}
		mes "ร้านนี้ใช้คะแนน Re-Option เท่านั้น";
		close2;
		callshop "ReOptionPointShop",1;
		end;
	default:
		close;
	}
	mes "[Re-Option]";
	mes "เลือกอุปกรณ์ที่สวมใส่";
	// ใช้ inventory index กันอุปกรณ์หลายตำแหน่งแสดงซ้ำ เช่น อาวุธสองมือ
	for (.@i = 0; .@i < getarraysize(.EquipSlots); ++.@i) {
		.@pos = .EquipSlots[.@i];
		.@idx = getequipinventoryindex(.@pos);
		if (.@idx < 0) continue;
		.@duplicate = 0;
		for (.@j = 0; .@j < .@count; ++.@j)
			if (.@indices[.@j] == .@idx) .@duplicate = 1;
		if (.@duplicate) continue;
		.@id = getequipid(.@pos);
		.@type = getiteminfo(.@id,ITEMINFO_TYPE);
		if (.@type != IT_ARMOR && .@type != IT_WEAPON) continue;
		if (!getequipisenableref(.@pos)) continue;
		.@refine = getequiprefinerycnt(.@pos);
		if (.@refine < 5 || getequiprandomoption(.@pos,0,ROA_ID) <= 0) continue;
		.@positions[.@count] = .@pos;
		.@indices[.@count] = .@idx;
		.@ids[.@count] = .@id;
		.@refines[.@count] = .@refine;
		.@uids$[.@count] = getequipuniqueid(.@pos);
		.@menu$ += "+" + .@refine + " " + getitemname(.@id) + ":";
		++.@count;
	}
	if (!.@count) {
		mes "ไม่มีอุปกรณ์ที่ผ่านเงื่อนไข";
		close;
	}
	.@selected = select(.@menu$ + "ยกเลิก") - 1;
	if (.@selected == .@count) close;
	.@pos = .@positions[.@selected];
	.@id = .@ids[.@selected];
	.@refine = .@refines[.@selected];
	.@row = -1;
	for (.@i = 0; .@i < getarraysize(.RefineLevels); ++.@i)
		if (.RefineLevels[.@i] == .@refine) .@row = .@i;
	// อาวุธใช้ subtype ส่วนชุดเกราะใช้รายการ Item ID จาก array เดิม
	if (getiteminfo(.@id,ITEMINFO_TYPE) == IT_WEAPON) {
		for (.@i = 0; .@i < getarraysize(.WeaponTypes); ++.@i)
			if (.WeaponTypes[.@i] == getiteminfo(.@id,ITEMINFO_SUBTYPE)) .@category$ = "Weapon";
	} else {
		for (.@i = 0; .@i < getarraysize(.RecipeCategories$); ++.@i) {
			.@name$ = .RecipeCategories$[.@i];
			for (.@j = 0; .@j < getarraysize(getd("." + .@name$ + "Items")); ++.@j)
				if (getd("." + .@name$ + "Items[" + .@j + "]") == .@id) .@category$ = .@name$;
		}
	}
	if (.@row < 0 || .@category$ == "") {
		mes "ยังไม่ได้กำหนดสูตรสำหรับอุปกรณ์นี้";
		close;
	}
	.@ticket = .ReOptionTicketIDs[.@row];
	.@ticket_amount = getd("." + .@category$ + "TicketAmounts[" + .@row + "]");
	.@mat1 = getd("." + .@category$ + "Material1IDs[" + .@row + "]");
	.@amount1 = getd("." + .@category$ + "Material1Amounts[" + .@row + "]");
	.@mat2 = getd("." + .@category$ + "Material2IDs[" + .@row + "]");
	.@amount2 = getd("." + .@category$ + "Material2Amounts[" + .@row + "]");
	.@zeny = getd("." + .@category$ + "ZenyCosts[" + .@row + "]");
	.@group = getequiprefinegroup(.@pos);
	if (.@ticket <= 0 || .@ticket_amount <= 0 || .@group <= 0) {
		mes "ยังไม่ได้ตั้งค่าใบ Re-Option หรือกลุ่มสุ่มของระดับนี้";
		mes "ไม่มีการหักค่าใช้จ่าย";
		close;
	}
	next;
	mes "[Re-Option]";
	mes "+" + .@refine + " " + mesitemlink(.@id);
	mes "วัตถุดิบที่ใช้";
	mes ""+ mesitemlink(.@ticket) + " x " + .@ticket_amount;
	if (.@amount1 > 0) mes mesitemlink(.@mat1) + " x " + .@amount1;
	if (.@amount2 > 0) mes mesitemlink(.@mat2) + " x " + .@amount2;
	mes "Zeny: " + F_InsertComma(.@zeny);
	mes "จะสุ่มใหม่ทั้งหมด (มีโอกาสได้ออฟชั่นเดิม)";
	if (select("ดำเนินการ:ยกเลิก") == 2) close;
	// ตรวจซ้ำและหักค่าใช้จ่ายในคำสั่งเดียว ไม่มีช่วงรอระหว่างหักกับสุ่ม
	.@result = rerollequiprefineoption(.@pos,.@indices[.@selected],.@uids$[.@selected],
		.@id,.@refine,.@ticket,.@ticket_amount,.@mat1,.@amount1,.@mat2,.@amount2,.@zeny);
	if (.@result == 1) {
		mes "เปลี่ยน Random Option เรียบร้อยแล้ว";
		.@optidna = getequiprandomoption(.@pos,0,ROA_ID);
		.@optvalna = getequiprandomoption(.@pos,0,ROA_VALUE);
		.@options$ += callfunc("F_GetRDMOPT_Name",.@optidna) + callfunc("F_GetRDMOPT_Value",.@optidna,.@optvalna);
		announce "["+strcharinfo(0)+"] รีออฟชั่น ["+ itemlink(.@id,.@refine,0,0,0,0,0,.@optidna,.@optvalna,.@optramamna)+"] ได้รับ [" + .@options$ + "]",bc_all;
	}
	else if (.@result == -1) mes "อุปกรณ์เปลี่ยนไปหรือไม่ผ่านเงื่อนไข กรุณาเลือกใหม่";
	else if (.@result == -2) mes "ข้อมูลสูตรหรือกลุ่มสุ่มไม่สมบูรณ์";
	else if (.@result == -3) mes "ไอเทมหรือ Zeny ไม่เพียงพอ";
	else mes "ไม่สามารถทำรายการขณะตายหรือเปิดร้านหรือแลกเปลี่ยนได้";
	close;

OnSellItem:
	// Freeze the selected items in local arrays before any mes/next/select can yield.
	if (!@ReOptionSellSession) end;
	@ReOptionSellSession = 0;
	.@count = getarraysize(@sold_nameid);
	if (.@count <= 0 || .@count > 100) end;
	for (.@i = 0; .@i < .@count; ++.@i) {
		.@recipe = inarray(.ExchangeIDs,@sold_nameid[.@i]);
		if (.@recipe < 0) {
			++.@skipped; // Not accepted: leave this inventory item untouched.
			continue;
		}
		.@n = .@accepted;
		.@idx[.@n] = @sold_idx[.@i];
		.@id[.@n] = @sold_nameid[.@i];
		.@quantity[.@n] = @sold_quantity[.@i];
		.@uid$[.@n] = @sold_uniqueid$[.@i];
		.@refine[.@n] = @sold_refine[.@i];
		.@attribute[.@n] = @sold_attribute[.@i];
		.@identify[.@n] = @sold_identify[.@i];
		.@grade[.@n] = @sold_enchantgrade[.@i];
		.@points[.@n] = .ExchangePoints[.@recipe] * .@quantity[.@n];
		if (.@quantity[.@n] <= 0 || .ExchangePoints[.@recipe] <= 0) goto L_SellReject;
		// Fixed fields: avoid nine script loop iterations per selected item.
		.@card1[.@n] = @sold_card1[.@i];
		.@card2[.@n] = @sold_card2[.@i];
		.@card3[.@n] = @sold_card3[.@i];
		.@card4[.@n] = @sold_card4[.@i];
		.@option_id1[.@n] = @sold_option_id1[.@i];
		.@option_value1[.@n] = @sold_option_val1[.@i];
		.@option_parameter1[.@n] = @sold_option_param1[.@i];
		.@option_id2[.@n] = @sold_option_id2[.@i];
		.@option_value2[.@n] = @sold_option_val2[.@i];
		.@option_parameter2[.@n] = @sold_option_param2[.@i];
		.@option_id3[.@n] = @sold_option_id3[.@i];
		.@option_value3[.@n] = @sold_option_val3[.@i];
		.@option_parameter3[.@n] = @sold_option_param3[.@i];
		.@option_id4[.@n] = @sold_option_id4[.@i];
		.@option_value4[.@n] = @sold_option_val4[.@i];
		.@option_parameter4[.@n] = @sold_option_param4[.@i];
		.@option_id5[.@n] = @sold_option_id5[.@i];
		.@option_value5[.@n] = @sold_option_val5[.@i];
		.@option_parameter5[.@n] = @sold_option_param5[.@i];
		.@gain += .@points[.@n];
		if (.@gain <= 0 || .@gain > .MaxPoints) goto L_SellReject;
		++.@accepted;
	}
	// The transaction now owns local snapshots, not reusable @sold_* shop variables.
	deletearray @sold_nameid;
	deletearray @sold_quantity;
	deletearray @sold_idx;
	deletearray @sold_uniqueid$;
	if (!.@accepted) {
		mes "[Re-Option]";
		mes "ไม่มีรายการที่อยู่ในลิสต์รับแลก";
		mes "ไม่ได้ลบไอเทมหรือเพิ่มคะแนน";
		close;
	}
	.@balance = #ReOptionPoints;
	// Pass 0 checks before preview; pass 1 rechecks after confirmation.
	for (.@pass = 0; .@pass < 2; ++.@pass) {
		// Build a direct lookup once per pass; +1 preserves inventory slot zero.
		getinventorylist;
		deletearray .@invlookup;
		for (.@j = 0; .@j < @inventorylist_count; ++.@j)
			.@invlookup[@inventorylist_idx[.@j]] = .@j + 1;
		deletearray .@seen;
		for (.@i = 0; .@i < .@accepted; ++.@i) {
			if (.@idx[.@i] < 0 || .@idx[.@i] >= MAX_INVENTORY) goto L_SellReject;
			if (.@seen[.@idx[.@i]]) goto L_SellReject;
			.@seen[.@idx[.@i]] = 1;
			.@inv = .@invlookup[.@idx[.@i]] - 1;
			if (.@inv < 0 || .@quantity[.@i] > @inventorylist_amount[.@inv]) goto L_SellReject;
			if (@inventorylist_id[.@inv] != .@id[.@i] ||
				@inventorylist_uniqueid$[.@inv] != .@uid$[.@i] ||
				@inventorylist_equip[.@inv] || @inventorylist_equipswitch[.@inv] ||
				@inventorylist_expire[.@inv] || @inventorylist_bound[.@inv] ||
				@inventorylist_refine[.@inv] != .@refine[.@i] ||
				@inventorylist_attribute[.@inv] != .@attribute[.@i] ||
				@inventorylist_identify[.@inv] != .@identify[.@i] ||
				@inventorylist_enchantgrade[.@inv] != .@grade[.@i] ||
				@inventorylist_card1[.@inv] != .@card1[.@i] ||
				@inventorylist_card2[.@inv] != .@card2[.@i] ||
				@inventorylist_card3[.@inv] != .@card3[.@i] ||
				@inventorylist_card4[.@inv] != .@card4[.@i] ||
				@inventorylist_option_id1[.@inv] != .@option_id1[.@i] ||
				@inventorylist_option_value1[.@inv] != .@option_value1[.@i] ||
				@inventorylist_option_parameter1[.@inv] != .@option_parameter1[.@i] ||
				@inventorylist_option_id2[.@inv] != .@option_id2[.@i] ||
				@inventorylist_option_value2[.@inv] != .@option_value2[.@i] ||
				@inventorylist_option_parameter2[.@inv] != .@option_parameter2[.@i] ||
				@inventorylist_option_id3[.@inv] != .@option_id3[.@i] ||
				@inventorylist_option_value3[.@inv] != .@option_value3[.@i] ||
				@inventorylist_option_parameter3[.@inv] != .@option_parameter3[.@i] ||
				@inventorylist_option_id4[.@inv] != .@option_id4[.@i] ||
				@inventorylist_option_value4[.@inv] != .@option_value4[.@i] ||
				@inventorylist_option_parameter4[.@inv] != .@option_parameter4[.@i] ||
				@inventorylist_option_id5[.@inv] != .@option_id5[.@i] ||
				@inventorylist_option_value5[.@inv] != .@option_value5[.@i] ||
				@inventorylist_option_parameter5[.@inv] != .@option_parameter5[.@i]) goto L_SellReject;
		}
		if (#ReOptionPoints < 0 || #ReOptionPoints > .MaxPoints - .@gain) goto L_SellReject;
		if (#ReOptionPoints != .@balance) goto L_SellBalanceChanged;
		if (.@pass == 0) {
			mes "[ยืนยันแลกคะแนน Re-Option]";
			for (.@i = 0; .@i < .@accepted; ++.@i) {
				mes "+" + .@refine[.@i] + " " + mesitemlink(.@id[.@i]) + " x " + .@quantity[.@i] + " = " + .@points[.@i] + " แต้ม";
			}
			mes "แลกทั้งหมด: " + .@accepted + " ชิ้น";
			mes "รวมเป็น: " + .@gain +" แต้ม";
			setdialogsize(350,400);
			next;
			if (select("ยืนยันแลก:ยกเลิก") == 2) {
				close;
			}
		}
	}
	// No yields after the final complete validation; delete only accepted snapshot indices.
	for (.@i = 0; .@i < .@accepted; ++.@i)
		delitemidx .@idx[.@i],.@quantity[.@i];
	#ReOptionPoints += .@gain;
	mes "แลกสำเร็จ ได้รับ " + .@gain + " แต้ม";
	mes "รวมเป็น: " + #ReOptionPoints +" แต้ม";
	close;

L_SellBalanceChanged:
	mes "[Re-Option]";
	mes "คะแนนเปลี่ยนระหว่างยืนยัน กรุณาเลือกทำรายการใหม่";
	mes "ไม่มีการลบไอเทมหรือเพิ่มคะแนน";
	close;

L_SellReject:
	deletearray @sold_nameid;
	deletearray @sold_quantity;
	deletearray @sold_idx;
	deletearray @sold_uniqueid$;
	mes "[Re-Option]";
	mes "รายการที่รับแลกมีข้อมูลเปลี่ยนไปหรือไม่ผ่านเงื่อนไข";
	mes "ไม่มีการลบไอเทมหรือเพิ่มคะแนน กรุณาเลือกใหม่";
	close;

	// การรีออฟชั่นตีบวกไอเท็มประเภท สวมใส่ - Equipment

	for (.@tier = 1; .@tier <= getarraysize(.PointValues); ++.@tier) {

			if (inarray(.ExchangeIDs,.@id) >= 0) {

				continue;

			continue;
		}

		npcshopadditem "ReOptionPointShop",.@id,.@price;
	}
