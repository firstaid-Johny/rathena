// Footprints require movement, so these are previewed on the player instead of statues.
prontera,208,181,4	script	Footprint Preview::HAT_FOOTPRINT_PREVIEW	4_M_SAGE_C,{
	mes "[Footprint Preview]";
	mes "Choose a footprint, then walk to see it.";
	mes "Selecting another footprint replaces only your previous preview.";
	.@choice = select(.menu$);
	if (.@choice == 1) {
		callsub L_ClearPreview;
		mes "Footprint preview stopped.";
		close;
	}
	if (.@choice == .count + 2)
		close;

	.@index = .@choice - 2;
	if (.@index < 0 || .@index >= .count)
		close;
	.@effect = .effects[.@index];
	if (@HatGalleryFootprint != .@effect || !has_hateffect(.@effect)) {
		callsub L_ClearPreview;
		// Never take ownership of an effect the player already has.
		if (!has_hateffect(.@effect)) {
			hateffect .@effect, true;
			@HatGalleryFootprintAdded = 1;
		}
		@HatGalleryFootprint = .@effect;
	}
	mes .names$[.@index];
	mes "Hat Effect ID: " + .@effect;
	mes "Walk a few steps. Return here to change or stop the preview.";
	close;

L_ClearPreview:
	if (@HatGalleryFootprint > 0 && @HatGalleryFootprintAdded) {
		if (has_hateffect(@HatGalleryFootprint))
			hateffect @HatGalleryFootprint, false;
		// Restore equipment effects if the same footprint was equipped during preview.
		recalculatestat;
	}
	@HatGalleryFootprint = 0;
	@HatGalleryFootprintAdded = 0;
	return;

OnInit:
	setarray .effects[0],
		FOOTPRINT_EF_STR_BASE,
		FOOTPRINT_EF_PURPLESTAR,
		FOOTPRINT_EF_YELLOWSTAR,
		FOOTPRINT_EF_REDSTAR,

		FOOTPRINT_EF_DRAGON_FACE_2D,
		FOOTPRINT_EF_DRAGON_FACE_3D,
		FOOTPRINT_EF_DIVINE,
		FOOTPRINT_EF_NYAR_BLUE,
		FOOTPRINT_EF_NYAR_PURPLE,
		FOOTPRINT_EF_DIVINE_BLUE,
		FOOTPRINT_EF_BLUESTAR,
		FOOTPRINT_EF_PHOENIX;
	setarray .names$[0],
		"FOOTPRINT_EF_STR_BASE",
		"FOOTPRINT_EF_PURPLESTAR",
		"FOOTPRINT_EF_YELLOWSTAR",
		"FOOTPRINT_EF_REDSTAR",
		"FOOTPRINT_EF_DOGFOOT",
