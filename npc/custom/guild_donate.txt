// Guild Donate: cumulative thresholds; highest tier only; no polling timer.
// Every valid DB tier applies its Script/Icon automatically.
prontera,160,181,4	script	Guild Donate	4_M_MANAGER,{
	mes "[Guild Donate]";
	.@guild = getcharid(2);
	if (!.@guild) {
		mes "คุณต้องเป็นสมาชิก Guild ก่อนบริจาค";
		close;
	}
	if (!guilddonaterefresh()) {
		mes "ระบบบริจาคไม่พร้อมใช้งาน กรุณาลองใหม่ภายหลัง";
		close;
	}
	.@total = guilddonateinfo(0);
	.@level = guilddonateinfo(1);
	mes "Guild: " + getguildname(.@guild);
	mes "ระดับบริจาค: " + .@level + " / " + guilddonateinfo(4);
	mes "ยอดสะสม: " + F_InsertComma(.@total) + " Zeny";
	mes "สามารถทยอยบริจาคได้ ยอดจะสะสมร่วมกันทั้ง Guild";
	if (.@level < guilddonateinfo(4))
		mes "อีก ^FF0000" + F_InsertComma(guilddonateinfo(2)) + "^000000 Zeny จะถึงขั้นถัดไป";
	setdialogsize(350,350);
	next;
	switch (select("บริจาค Zeny:ดูยอดที่ต้องใช้แต่ละขั้น:ยกเลิก")) {
	case 1:
		if (.@level >= guilddonateinfo(4)) {
			mes "Guild ของคุณถึงขั้นสูงสุดแล้ว";
			close;
		}
		if (Zeny < 1) {
			mes "คุณมี Zeny ไม่เพียงพอ";
			close;
		}
		mes "ระบุจำนวน Zeny ที่ต้องการบริจาค";
		if (input(.@amount,1,Zeny) != 0) {
			mes "จำนวนเงินไม่ถูกต้อง กรุณาทำรายการใหม่";
			close;
		}
		mes "ยืนยันบริจาค " + F_InsertComma(.@amount) + " Zeny ให้ Guild นี้?";
		mes "หากเกินยอดขั้นสูงสุด จะหักเฉพาะจำนวนที่ยังขาด";
		next;
		if (select("ยืนยัน:ยกเลิก") == 2) close;
		.@paid = guilddonate(.@amount,.@guild);
		mes "[Guild Donate]";
		if (.@paid > 0) {
			mes "บริจาคสำเร็จ: " + F_InsertComma(.@paid) + " Zeny";
			mes "ยอดสะสม: " + F_InsertComma(guilddonateinfo(0)) + " Zeny";
			mes "ระดับบริจาค: " + guilddonateinfo(1);
		} else if (.@paid == -1) mes "ข้อมูลสมาชิก Guild เปลี่ยนแปลง กรุณาคุยใหม่อีกครั้ง";
		else if (.@paid == -2) mes "ระบบบริจาคปิดอยู่หรือไม่พร้อมใช้งาน";
		else if (.@paid == -3) mes "จำนวนไม่ถูกต้องหรือ Zeny ไม่เพียงพอ";
		else if (.@paid == -4) mes "Guild ถึงขั้นสูงสุดแล้ว";
		else if (.@paid == -6) mes "ไม่สามารถบริจาคขณะทำรายการอื่นได้ กรุณาลองใหม่";
		else mes "บันทึกข้อมูลไม่สำเร็จ กรุณาลองใหม่ภายหลัง";
		close;
	case 2:
		mes "[ยอดสะสมแต่ละขั้น]";
		for (.@i = 0; .@i < guilddonateinfo(4); ++.@i) {
			if (.@i && .@i % 10 == 0) { next; mes "[ยอดสะสมแต่ละขั้น]"; }
			mes "ขั้น " + (.@i+1) + ": " + F_InsertComma(guilddonateinfo(5,.@i+1)) + " Zeny";
		}
		close;
	default:
		close;
	}
OnInit:
	waitingroom "Guild Donation",0;
	// Levels and cumulative RequiredZeny are read from guild_donate_db.yml.
	.Ready = guilddonateconfigure();
	if (!.Ready) debugmes "Guild Donate: invalid DB tiers/RequiredZeny or SQL is unavailable.";
	end;
}
